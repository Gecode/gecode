cmake_minimum_required(VERSION 3.21)
project(gecode_allocator_consumer LANGUAGES CXX)
enable_testing()
find_package(Gecode CONFIG REQUIRED COMPONENTS kernel)
find_package(Threads REQUIRED)

set(GECODE_LINK_VARIANT shared CACHE STRING "Installed Gecode variant: shared or static")
option(GECODE_EXPECT_MIMALLOC "Require the mimalloc backend" OFF)
set(MIMALLOC_INCLUDE_DIR "" CACHE PATH "Private source headers for the static ownership check")
if(GECODE_LINK_VARIANT STREQUAL "shared")
  add_library(allocator-boundary SHARED boundary.cpp)
elseif(GECODE_LINK_VARIANT STREQUAL "static")
  add_library(allocator-boundary STATIC boundary.cpp)
  target_compile_definitions(allocator-boundary PUBLIC ALLOCATOR_BOUNDARY_STATIC)
else()
  message(FATAL_ERROR "GECODE_LINK_VARIANT must be shared or static")
endif()
target_compile_definitions(allocator-boundary PRIVATE ALLOCATOR_BOUNDARY_BUILD)
target_link_libraries(allocator-boundary PUBLIC Gecode::gecodekernel_${GECODE_LINK_VARIANT})
add_executable(allocator-consumer main.cpp)
target_link_libraries(allocator-consumer PRIVATE allocator-boundary Threads::Threads)
target_compile_definitions(allocator-consumer PRIVATE GECODE_EXPECT_MIMALLOC=$<BOOL:${GECODE_EXPECT_MIMALLOC}>)
if(GECODE_EXPECT_MIMALLOC AND GECODE_LINK_VARIANT STREQUAL "static")
  if(NOT EXISTS "${MIMALLOC_INCLUDE_DIR}/mimalloc.h")
    message(FATAL_ERROR "The static ownership check requires MIMALLOC_INCLUDE_DIR")
  endif()
  target_compile_definitions(allocator-consumer PRIVATE CHECK_MIMALLOC_OWNERSHIP)
  target_include_directories(allocator-consumer PRIVATE "${MIMALLOC_INCLUDE_DIR}")
endif()
add_test(NAME allocator-consumer COMMAND allocator-consumer)
